function drawImageCover(
  ctx,
  image,
  destX,
  destY,
  destWidth,
  destHeight
) {
  const sourceRatio =
    image.width / image.height;

  const destRatio =
    destWidth / destHeight;

  let sx = 0;
  let sy = 0;
  let sw = image.width;
  let sh = image.height;

  if (sourceRatio > destRatio) {
    sw =
      image.height * destRatio;

    sx =
      (image.width - sw) / 2;
  } else {
    sh =
      image.width / destRatio;

    sy =
      (image.height - sh) / 2;
  }

  ctx.drawImage(
    image,
    sx,
    sy,
    sw,
    sh,
    destX,
    destY,
    destWidth,
    destHeight
  );
}

Base FOV = 1 means:
